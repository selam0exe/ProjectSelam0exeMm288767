Hub.CoinFarm.MoveToCoin = function(arg, arg2, arg3)
	local mode = Hub.CoinFarm.mode
	local position = Hub.CoinFarm.position
	local escapeUnderActive = Hub.CoinFarm.escapeCoin == arg3
	local str = escapeUnderActive and "Under" or position
	Hub.CoinFarm.escapeUnderActive = escapeUnderActive
	local position2 = arg2.Position
	local v8 = Hub.CoinFarm.GetTargetPosition(arg, arg2, arg3, str)
	local magnitude = (v8 - position2).Magnitude
	local speed
	if mode == "Tween" then
		speed = Hub.CoinFarm.tweenSpeed or Hub.CoinFarm.speed or 28
	else
		speed = Hub.CoinFarm.speed or 28
	end
	local n2 = math.max(magnitude / math.max(1, speed), 0.01)

	if str == "Under" or mode == "Tween" then
		Hub.CoinFarm.EnableUnder(arg, false, escapeUnderActive)
	else
		Hub.CoinFarm.DisableUnder()
	end

	if mode == "Tween" then
		Hub.CoinFarm.CancelTween()
		local tween = TweenService:Create(arg2, TweenInfo.new(n2, Enum.EasingStyle.Linear, Enum.EasingDirection.Out), { CFrame = CFrame.new(v8) })
		Hub.CoinFarm.activeTween = tween
		tween:Play()

		while tween.PlaybackState == Enum.PlaybackState.Playing do
			if not Hub.CoinFarm.IsTravelValid(arg2, arg3, mode, position) then
				tween:Cancel()
				break
			else
				local v9 = Hub.CoinFarm.GetNearbyMurderer(arg2)

				if v9 and Hub.CoinFarm.EscapeMurderer(arg2, v9) then
					tween:Cancel()
					break
				else
					if str == "Under" or mode == "Tween" then
						Hub.CoinFarm.EnableUnder(arg, false, escapeUnderActive)
					end

					RunService.Heartbeat:Wait()
				end
			end
		end

		local flag7 = tween.PlaybackState == Enum.PlaybackState.Completed

		if Hub.CoinFarm.activeTween == tween then
			Hub.CoinFarm.activeTween = nil
		end

		pcall(function()
			tween:Destroy()
		end)

		if flag7 and str == "Under" then
			Hub.CoinFarm.EnableUnder(arg, true, escapeUnderActive)
		end

		return flag7
	end

	local now = os.clock()
	local v9 = now

	while Hub.CoinFarm.IsTravelValid(arg2, arg3, mode, position) do
		local now2 = os.clock()
		local n3 = math.clamp((now2 - now) / n2, 0, 1)

		if str == "Under" then
			Hub.CoinFarm.EnableUnder(arg, false, escapeUnderActive)
		end

		local v10 = Hub.CoinFarm.GetNearbyMurderer(arg2)
		if v10 and Hub.CoinFarm.EscapeMurderer(arg2, v10) then
			return false
		end
		arg2.AssemblyLinearVelocity = Vector3.zero
		arg2.AssemblyAngularVelocity = Vector3.zero
		local n4 = position2:Lerp(v8, n3)
		local n5 = math.max(3, speed * math.min(now2 - v9, 0.12) * 1.35)
		local n6 = n4 - arg2.Position

		if n5 < n6.Magnitude then
			n4 = arg2.Position + n6.Unit * n5
		end

		arg2.CFrame = CFrame.new(n4)

		if n3 >= 1 then
			if str == "Under" then
				Hub.CoinFarm.EnableUnder(arg, true, escapeUnderActive)
			end

			return true
		end

		RunService.Heartbeat:Wait()
		v9 = now2
	end

	return false
end

Hub.CoinFarm.Worker = function()
	if Hub.CoinFarm.workerRunning then
		return
	end
	Hub.CoinFarm.workerRunning = true
	local generation = Hub.CoinFarm.generation

	task.spawn(function()
		while foxnameMM2Runtime.alive and not isTerminated and Hub.CoinFarm.enabled and Hub.CoinFarm.generation == generation do
			local character_ = localPlayer.Character
			local humanoid = character_ and character_:FindFirstChildOfClass("Humanoid")
			local humanoidRootPart = character_ and character_:FindFirstChild("HumanoidRootPart")

			if not character_ or not humanoid or humanoid.Health <= 0 or not humanoidRootPart then
				Hub.CoinFarm.SetPhase("Waiting for character")
				RunService.Heartbeat:Wait()
				continue
			end

			if not Hub.RoundState.active then
				Hub.CoinFarm.SetPhase("Waiting for round")
				RunService.Heartbeat:Wait()
				continue
			end

			if not Hub.RoundState.coinsReady then
				Hub.CoinFarm.SetPhase("Waiting for coins")
				RunService.Heartbeat:Wait()
				continue
			end

			if Hub.CoinFarm.bagFull then
				Hub.CoinFarm.SetPhase("Coin bag full")
				RunService.Heartbeat:Wait()
				continue
			end

			if Hub.CoinFarm.sheriffFlingBlocking then
				Hub.CoinFarm.SetPhase("Ending round: Fling")
				RunService.Heartbeat:Wait()
				continue
			end

			if Hub.FlingActive then
				RunService.Heartbeat:Wait()
				continue
			end

			if not Hub.CoinFarm.IsLocalPlayerInRound() then
				Hub.CoinFarm.SetPhase("Waiting for round role")
				RunService.Heartbeat:Wait()
				continue
			end

			if not Hub.CoinFarm.safetyChecked then
				if not Hub.CoinFarm.RunInitialSafety(humanoidRootPart) then
					Hub.CoinFarm.SetPhase("Safety check")
					RunService.Heartbeat:Wait()
					continue
				end
			end

			if Hub.CoinFarm.CheckBagFullGui() then
				RunService.Heartbeat:Wait()
				continue
			end

			if not Hub.CoinFarm.IndexMap() then
				Hub.CoinFarm.SetPhase("Indexing map")
				task.wait(0.1)
				continue
			end

			local escapeCoin = Hub.CoinFarm.escapeCoin
			local currentCoin

			if escapeCoin and Hub.CoinFarm.IsCoinAvailable(escapeCoin) then
				currentCoin = escapeCoin
			else
				Hub.CoinFarm.escapeCoin = nil
				Hub.CoinFarm.escapeUnderActive = false
				currentCoin = Hub.CoinFarm.currentCoin

				if not currentCoin or not Hub.CoinFarm.IsCoinAvailable(currentCoin) then
					local now = os.clock()
					local retryUntil = currentCoin and Hub.CoinFarm.retryUntil[currentCoin]

					if retryUntil and now < retryUntil then
						-- still retrying
					else
						currentCoin = Hub.CoinFarm.GetNearestCoin(humanoidRootPart)
						Hub.CoinFarm.currentCoin = currentCoin
					end
				end
			end

			if not currentCoin then
				Hub.CoinFarm.SetPhase("No coins found")
				task.wait(0.15)
				continue
			end

			Hub.CoinFarm.ranThisRound = true
			Hub.CoinFarm.SetPhase("Farming")

			local moved = Hub.CoinFarm.MoveToCoin(character_, humanoidRootPart, currentCoin)

			if not (foxnameMM2Runtime.alive and not isTerminated and Hub.CoinFarm.enabled and Hub.CoinFarm.generation == generation) then
				break
			end

			if moved then
				local attempts = (Hub.CoinFarm.contactAttempts[currentCoin] or 0) + 1
				Hub.CoinFarm.contactAttempts[currentCoin] = attempts

				local touched = Hub.CoinFarm.TouchCoin(character_, humanoidRootPart, currentCoin)

				if touched then
					task.wait(0.05)

					if not Hub.CoinFarm.IsCoinAvailable(currentCoin) then
						Hub.CoinFarm.MarkCurrentCollected()
					else
						Hub.CoinFarm.retryUntil[currentCoin] = os.clock() + 0.3
						Hub.CoinFarm.currentCoin = nil
					end
				else
					Hub.CoinFarm.retryUntil[currentCoin] = os.clock() + 0.5
					Hub.CoinFarm.currentCoin = nil
				end
			else
				if Hub.CoinFarm.currentCoin == currentCoin then
					local attempts = (Hub.CoinFarm.contactAttempts[currentCoin] or 0) + 1
					Hub.CoinFarm.contactAttempts[currentCoin] = attempts

					if attempts >= 5 then
						Hub.CoinFarm.coins[currentCoin] = nil
						Hub.CoinFarm.retryUntil[currentCoin] = nil
						Hub.CoinFarm.contactAttempts[currentCoin] = nil
						Hub.CoinFarm.currentCoin = nil
					else
						Hub.CoinFarm.retryUntil[currentCoin] = os.clock() + 0.4
						Hub.CoinFarm.currentCoin = nil
					end
				end
			end

			RunService.Heartbeat:Wait()
		end

		Hub.CoinFarm.workerRunning = false
	end)
end

Hub.CoinFarm.FindOwnedCoinAmountLabels = function()
	local results = {}
	local playerGui = localPlayer:FindFirstChild("PlayerGui")
	if not playerGui then
		return results
	end

	local function search(parent)
		for _, child in ipairs(parent:GetChildren()) do
			if child:IsA("TextLabel") then
				local text = child.Text or ""
				if text:match("^%d") or text:match("Coin") or text:match("coin") then
					table.insert(results, child)
				end
			end
			search(child)
		end
	end

	local mainGui = playerGui:FindFirstChild("MainGUI")
	if mainGui then
		search(mainGui)
	end

	return results
end

Hub.TouchButtons.LoadRemoteIcon = function(button, iconFile, title)
	if not button or not button.Parent then
		return
	end
	local remoteCache = Hub.TouchButtons.remoteIconCache
	local remoteLoading = Hub.TouchButtons.remoteIconLoading
	local remoteFailed = Hub.TouchButtons.remoteIconFailed

	if remoteCache[iconFile] then
		pcall(function()
			local img = button:FindFirstChildWhichIsA("ImageLabel")
			if img then
				img.Image = remoteCache[iconFile]
				img.Visible = true
			end
		end)
		return
	end

	if remoteFailed[iconFile] then
		local lucideIcon = Hub.TouchButtons.Icons[title]
		if lucideIcon then
			Hub.TouchButtons.ApplyLucideIcon(button, lucideIcon)
		end
		return
	end

	if remoteLoading[iconFile] then
		return
	end

	remoteLoading[iconFile] = true

	task.spawn(function()
		local url = "https://raw.githubusercontent.com/lo6ix/mm2-icons/main/" .. iconFile
		local ok, result = pcall(function()
			return game:HttpGet(url)
		end)

		if ok and type(result) == "string" and #result > 10 then
			remoteCache[iconFile] = result
			remoteLoading[iconFile] = nil

			if button and button.Parent then
				pcall(function()
					local img = button:FindFirstChildWhichIsA("ImageLabel")
					if img then
						img.Image = result
						img.Visible = true
					end
				end)
			end
		else
			remoteLoading[iconFile] = nil
			remoteFailed[iconFile] = true
			local lucideIcon = Hub.TouchButtons.Icons[title]
			if lucideIcon and button and button.Parent then
				Hub.TouchButtons.ApplyLucideIcon(button, lucideIcon)
			end
		end
	end)
end

Hub.TouchButtons.BindHoldDrag = function(button, state, gui)
	if not button or not state then
		return
	end

	if state.beginConn then
		pcall(function() state.beginConn:Disconnect() end)
		state.beginConn = nil
	end
	if state.inputConn then
		pcall(function() state.inputConn:Disconnect() end)
		state.inputConn = nil
	end
	if state.globalConn then
		pcall(function() state.globalConn:Disconnect() end)
		state.globalConn = nil
	end
	if state.endConn then
		pcall(function() state.endConn:Disconnect() end)
		state.endConn = nil
	end

	state.beginConn = button.InputBegan:Connect(function(input)
		if input.UserInputType ~= Enum.UserInputType.Touch and input.UserInputType ~= Enum.UserInputType.MouseButton1 then
			return
		end
		state.input = input
		state.start = os.clock()
		state.startPos = button.AbsolutePosition
		state.dragging = false
		state.dragMoved = false

		local holdThreshold = 0.35
		local startTime = state.start
		local startInput = input

		task.delay(holdThreshold, function()
			if state.input == startInput and state.start == startTime and not state.dragging and state.draggable then
				state.dragging = true
			end
		end)
	end)

	state.globalConn = UserInputService.InputChanged:Connect(function(input)
		if not state.dragging or state.input == nil then
			return
		end
		if input.UserInputType ~= Enum.UserInputType.Touch and input.UserInputType ~= Enum.UserInputType.MouseMovement then
			return
		end
		local delta = input.Position - state.input.Position
		local newPos = UDim2.new(
			0, state.startPos.X + delta.X,
			0, state.startPos.Y + delta.Y
		)
		local vp = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(800, 600)
		local bSize = button.AbsoluteSize
		local clampedX = math.clamp(newPos.X.Offset, 0, vp.X - bSize.X)
		local clampedY = math.clamp(newPos.Y.Offset, 0, vp.Y - bSize.Y)
		local relX = clampedX / vp.X
		local relY = clampedY / vp.Y
		state.position = UDim2.new(relX, 0, relY, 0)
		if gui and gui.Parent then
			gui.Position = state.position
		end
		state.dragMoved = true
	end)

	state.endConn = UserInputService.InputEnded:Connect(function(input)
		if input ~= state.input then
			return
		end
		state.dragging = false
		state.input = nil
	end)
end

Hub.TouchButtons.Disconnect = function(state)
	if not state then return end
	for _, key in ipairs({"beginConn","inputConn","globalConn","endConn"}) do
		if state[key] then
			pcall(function() state[key]:Disconnect() end)
			state[key] = nil
		end

end

Hub.TouchButtons.Destroy = function(state)
	if not state then return end
	Hub.TouchButtons.Disconnect(state)
	if state.gui and state.gui.Parent then
		pcall(function() state.gui:Destroy() end)
	end
	state.gui = nil
	state.button = nil
	state.enabled = false
	state.dragging = false
	state.dragMoved = false
	state.input = nil
end

Hub.TouchButtons.Create = function(state, title, onActivate, sizeKey, position)
	Hub.TouchButtons.Destroy(state)

	local gui = createScreenGui("FoxnameButton_" .. (title or "Btn"), 20)
	gui.Position = state.position or position or UDim2.new(0.84, 0, 0.5, 0)

	local sizeData = Hub.TouchButtons.Sizes[sizeKey or state.size or "Medium"] or Hub.TouchButtons.Sizes.Medium

	local button = Instance.new("TextButton")
	button.Name = title or "ActionButton"
	button.Size = sizeData.size
	button.Position = UDim2.new(0, 0, 0, 0)
	button.BackgroundColor3 = state.backgroundColor or Hub.TouchButtons.Theme.backgroundColor
	button.AutoButtonColor = false
	button.Text = ""
	button.Parent = gui

	Hub.TouchButtons.DecorateActionButton(button, title, state)
	Hub.TouchButtons.UpdateHoldHint(button, state.draggable)

	local lucideIcon = Hub.TouchButtons.Icons[title]
	if lucideIcon then
		Hub.TouchButtons.ApplyLucideIcon(button, lucideIcon)
	end

	gui.Position = state.position or position or UDim2.new(0.84, 0, 0.5, 0)

	state.gui = gui
	state.button = button
	state.dragging = false
	state.dragMoved = false
	state.input = nil

	Hub.TouchButtons.BindHoldDrag(button, state, gui)

	button.Activated:Connect(function()
		if Hub.TouchButtons.ShouldSuppressActivation(state) then
			return
		end
		if type(onActivate) == "function" then
			pcall(onActivate)
		end
	end)

	return gui, button
end

	end

	local payload = HttpService:JSONEncode({
		embeds = {

				description = tostring(description or ""),
				color = tonumber(color) or 10701977,

	local ok = pcall(function()
		request_fn({
			Url = url,
			Method = "POST",
			Headers = { ["Content-Type"] = "application/json" },
			Body = payload,

Hub.CoinFarm.FetchServerCandidates = function()
	local ok, result = pcall(function()
		return HttpService:JSONDecode(
			game:HttpGet(
				"https://games.roblox.com/v1/games/" .. tostring(game.PlaceId) .. "/servers/Public?sortOrder=Asc&limit=100"
			)
		)
	end)

	if not ok or type(result) ~= "table" or type(result.data) ~= "table" then
		return nil
	end

	local candidates = {}
	for _, server in ipairs(result.data) do
		if type(server.id) == "string" and server.id ~= game.JobId then
			local playing = tonumber(server.playing) or 0
			local maxPlayers = tonumber(server.maxPlayers) or 10
			if playing < maxPlayers then
				table.insert(candidates, { id = server.id, playing = playing, maxPlayers = maxPlayers })
			end
		end
	end

	table.sort(candidates, function(a, b)
		return a.playing < b.playing
	end)

	return candidates
end

Hub.CoinFarm.StartLowPlayerServerHop = function()
	if Hub.CoinFarm.serverHopRunning then
		return false
	end
	Hub.CoinFarm.serverHopRunning = true
	Hub.CoinFarm.serverHopCount = Hub.CoinFarm.serverHopCount + 1

	task.spawn(function()
		Hub.CoinFarm.ReportServerHop("Low player count")

		local candidates = Hub.CoinFarm.FetchServerCandidates()
		if not candidates or #candidates == 0 then
			Hub.CoinFarm.serverHopRunning = false
			return
		end

		local threshold = Hub.CoinFarm.serverHopPlayerThreshold or 4
		local target = nil

		for _, server in ipairs(candidates) do
			if server.playing <= threshold then
				target = server
				break
			end
		end

		if not target then
			target = candidates[1]
		end

		if target then
			Hub.CoinFarm.TryServerTeleport(target.id)
		else
			Hub.CoinFarm.serverHopRunning = false
		end
	end)

	return true
end

Hub.CoinFarm.CountLikelyBotPlayers = function()
	local count = 0
	local botPatterns = { "^%u%l+%d+$", "^Player%d+$", "^Guest%d+$" }

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= localPlayer then
			local name = player.Name or ""
			local isBot = false
			for _, pattern in ipairs(botPatterns) do
				if name:match(pattern) then
					isBot = true
					break
				end
			end
			if isBot then
				count = count + 1
			end
		end
	end

	return count
end

Hub.RunBombJump = function()
	if Hub.BombJump.busy then
		return false
	end

	local now = os.clock()
	if now < Hub.BombJump.nextReadyAt then
		local remaining = math.ceil(Hub.BombJump.nextReadyAt - now)
		WindUI:Notify({
			Title = "Bomb Jump",
			Content = "Ready in " .. tostring(remaining) .. "s",
			Icon = "clock",
			Duration = 1.5,
		})
		return false
	end

	local character_ = localPlayer.Character
	local humanoid = character_ and character_:FindFirstChildOfClass("Humanoid")
	local humanoidRootPart = character_ and character_:FindFirstChild("HumanoidRootPart")

	if not character_ or not humanoid or humanoid.Health <= 0 or not humanoidRootPart then
		return false
	end

	local backpack = localPlayer:FindFirstChildOfClass("Backpack")
	local bomb = character_:FindFirstChild("Bomb") or (backpack and backpack:FindFirstChild("Bomb"))

	if not bomb then
		WindUI:Notify({
			Title = "Bomb Jump",
			Content = "No bomb found",
			Icon = "package-x",
			Duration = 2,
		})
		return false
	end

	Hub.BombJump.busy = true
	Hub.BombJump.nextReadyAt = now + Hub.BombJump.cooldown

	task.spawn(function()
		local ok = pcall(function()
			local handle = bomb:FindFirstChild("Handle") or bomb:FindFirstChildWhichIsA("BasePart")
			if not handle then return end

			humanoidRootPart.AssemblyLinearVelocity = Vector3.new(0, 80, 0)

			if type(firetouchinterest) == "function" then
				firetouchinterest(humanoidRootPart, handle, 0)
				task.wait(0.05)
				firetouchinterest(humanoidRootPart, handle, 1)
			end
		end)

		task.wait(0.5)
		Hub.BombJump.busy = false
	end)

	return true
end

local function fn35()
	Hub.XRayScanGeneration = Hub.XRayScanGeneration + 1
	local gen = Hub.XRayScanGeneration
	Hub.XRayOriginal = {}
	Hub.XRayMap = nil

	local map = Hub.CoinFarm.ResolveMap() or workspace

	local function processDescendant(obj)
		if gen ~= Hub.XRayScanGeneration then return end
		if obj:IsA("BasePart") or obj:IsA("UnionOperation") or obj:IsA("MeshPart") or obj:IsA("SpecialMesh") then
			local chr = localPlayer.Character
			if chr and obj:IsDescendantOf(chr) then return end
			local isPlayer = false
			for _, p in ipairs(Players:GetPlayers()) do
				if p.Character and obj:IsDescendantOf(p.Character) then
					isPlayer = true
					break
				end
			end
			if isPlayer then return end
			if obj:IsA("BasePart") or obj:IsA("UnionOperation") or obj:IsA("MeshPart") then
				Hub.XRayOriginal[obj] = obj.LocalTransparencyModifier
			end
		end
	end

	for _, desc in ipairs(map:GetDescendants()) do
		processDescendant(desc)
	end

	Hub.XRayMap = map
end

local function getAimTarget()
local cam = workspace.CurrentCamera
if not cam then return nil end
local vp = cam.ViewportSize
local center = Vector2.new(vp.X / 2, vp.Y / 2)
local best, bestDist = nil, math.huge
for _, player in ipairs(Players:GetPlayers()) do
if player == localPlayer then continue end
if Hub.OnlyAimAsSheriff and fn27_getRole(localPlayer) ~= "Sheriff" then continue end
local chr = player.Character
local humanoid = chr and chr:FindFirstChildOfClass("Humanoid")
if not chr or not humanoid or humanoid.Health <= 0 then continue end
local partName = Hub.AimPart or "Head"
local part = chr:FindFirstChild(partName) or chr:FindFirstChild("HumanoidRootPart")
if not part then continue end
if not Hub.IgnoreWalls then
local origin = cam.CFrame.Position
local direction = (part.Position - origin)
local ray = workspace:Raycast(origin, direction, RaycastParams.new())
if ray and ray.Instance then
local inChar = false
for _, p in ipairs(Players:GetPlayers()) do
if p.Character and ray.Instance:IsDescendantOf(p.Character) then
inChar = true
break
end
end
if not inChar then continue end
end
end
local screenPos, onScreen = cam:WorldToViewportPoint(part.Position)
if not onScreen then continue end
local screenVec = Vector2.new(screenPos.X, screenPos.Y)
local dist = (screenVec - center).Magnitude
if dist < Hub.AimFOV and dist < bestDist then
best = part
bestDist = dist
end
end
return best
end

local predictedPos = target.Position + velocity * prediction
cam.CFrame = CFrame.new(cam.CFrame.Position, predictedPos)
end

end)
